#   python -m venv .venv-eval && source .venv-eval/bin/activate
#   pip install -r requirements.txt -r requirements-eval.txt
#   # 아래 핀이 운영 핀을 덮어쓴다 (langchain 계열 다운그레이드)

ragas==0.1.21

#   datasets>=3 : pyarrow가 numpy>=2 요구 → ragas가 고정한 numpy 1.26과 충돌
langchain==0.2.17

numpy==1.26.4

# 생성·판정 모델 호출. 운영과 같은 메이저를 유지해야 측정 조건이 같다.
# (ragas는 openai>=1.0.0만 요구하지만 전이 의존성이 2.x로 끌어올린다)
openai>=1.99.9,<2.0.0
